Total Revenue = SUM(Fact_Orders[sales_amount])
Total Profit = SUM(Fact_Orders[profit])
Total Orders = DISTINCTCOUNT(Fact_Orders[order_id])
Total Customers = DISTINCTCOUNT(Fact_Orders[customer_id])
AOV = DIVIDE([Total Revenue],[Total Orders])
Profit Margin % = DIVIDE([Total Profit],[Total Revenue])
Returned Orders = CALCULATE([Total Orders],Fact_Orders[return_flag]=1)
Return Rate % = DIVIDE([Returned Orders],[Total Orders])
Average Discount % = DIVIDE(AVERAGE(Fact_Orders[discount_pct]),100)
Revenue PY = CALCULATE([Total Revenue],SAMEPERIODLASTYEAR(Dim_Date[full_date]))
Revenue YoY % = DIVIDE([Total Revenue]-[Revenue PY],[Revenue PY])
Profit PY = CALCULATE([Total Profit],SAMEPERIODLASTYEAR(Dim_Date[full_date]))
Profit YoY % = DIVIDE([Total Profit]-[Profit PY],[Profit PY])
Revenue MoM % =
VAR Prev = CALCULATE([Total Revenue],DATEADD(Dim_Date[full_date],-1,MONTH))
RETURN DIVIDE([Total Revenue]-Prev,Prev)
Revenue per Customer = DIVIDE([Total Revenue],[Total Customers])
Profit per Order = DIVIDE([Total Profit],[Total Orders])
Return Revenue = CALCULATE([Total Revenue],Fact_Orders[return_flag]=1)
Return Profit Impact = CALCULATE([Total Profit],Fact_Orders[return_flag]=1)
Revenue Contribution % =
DIVIDE([Total Revenue],CALCULATE([Total Revenue],ALL(Dim_Product)))
